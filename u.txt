#!/bin/bash

echo "=== удаление пользователей, групп, файлов и настроек ==="

for u in u1 u2 myuser; do
if id $u &>/dev/null; then
userdel -r $u
echo "пользователь $u удалён"
fi
done

for g in g1 test13group; do
if getent group $g &>/dev/null; then
groupdel $g
echo "группа $g удалена"
fi
done

rm -rf /home/test13 /home/test14 /home/test15
echo "каталоги удалены"

rm -f /etc/skel/readme.txt
echo "файл readme.txt удалён"

rm -f /etc/sudoers.d/u1-passwd
echo "правила для u1 удалены"

rm -f work3.log
echo "лог-файл work3.log удалён"

echo "очистка завершена."
